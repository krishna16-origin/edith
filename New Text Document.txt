import json
import os
import re
from typing import Dict, List, Literal
from pydantic import BaseModel, Field
from langchain_community.tools.tavily_search import TavilySearchResults
from langchain_core.prompts import ChatPromptTemplate
from langchain_groq import ChatGroq
from langgraph.graph import END, StateGraph
import streamlit as st
import pandas as pd
import plotly.express as px
import plotly.graph_objects as go

# ==========================================
# ENVIRONMENT SETUP
# ==========================================
GROQ_API_KEY = os.getenv("GROQ_API_KEY")
TAVILY_API_KEY = os.getenv("TAVILY_API_KEY")

class SearchQueries(BaseModel):
    queries: List[str] = Field(description="List of search queries.")

class AgentState(BaseModel):
    question: str
    queries: List[str] = []
    search_results: List[Dict] = []
    iterations: int = 0
    final_answer: str = ""
    references: Dict[int, Dict] = {}

# Initialize Models and Tools
fast = ChatGroq(model="openai/gpt-oss-120b", temperature=0.1, api_key=GROQ_API_KEY)
reasoning = ChatGroq(model="openai/gpt-oss-120b", temperature=0.2, api_key=GROQ_API_KEY)
search_tool = TavilySearchResults(max_results=5, tavily_api_key=TAVILY_API_KEY)

# ==========================================
# GRAPH NODES
# ==========================================

def plan_research_node(state: AgentState) -> Dict:
    prompt = ChatPromptTemplate.from_messages([
        ("system", "Return ONLY JSON: {{\"queries\": [\"query1\", \"query2\"]}}"),
        ("human", "Generate 3 distinct search queries to thoroughly research: {question}")
    ])
    planner = prompt | fast
    response = planner.invoke({"question": state.question})
    try:
        result = json.loads(response.content)
        queries = result.get("queries", [state.question])
    except:
        queries = [state.question, state.question + " statistics data", state.question + " market trends"]
    return {"queries": queries, "iterations": 1}

def execute_search_node(state: AgentState) -> Dict:
    batch_results = []
    for query in state.queries:
        try:
            results = search_tool.invoke({"query": query})
            batch_results.append(results)
        except:
            batch_results.append([])
    
    seen_urls = {res["url"] for res in state.search_results}
    new_results = list(state.search_results)
    
    for results in batch_results:
        for item in results:
            if item.get("url") not in seen_urls:
                seen_urls.add(item["url"])
                new_results.append(item)
    
    return {"search_results": new_results}

def evaluate_research_node(state: AgentState) -> Literal["finalize", "continue"]:
    if state.iterations >= 3 or len(state.search_results) >= 12:
        return "finalize"
    return "continue"

def continue_research_node(state: AgentState) -> Dict:
    return {
        "queries": [state.question + " data analytics empirical statistics figures"],
        "iterations": state.iterations + 1
    }

def finalize_answer_node(state: AgentState) -> Dict:
    formatted_context = []
    references = {}
    
    for idx, res in enumerate(state.search_results[:15], start=1):
        references[idx] = {"title": res.get("title", f"Source {idx}"), "url": res.get("url", "")}
        content = res.get("content", "")[:1200] 
        formatted_context.append(f"SOURCE [{idx}]\nTitle: {res.get('title', '')}\nContent: {content}\n")
    
    context_str = "\n---\n".join(formatted_context)
    
    system_instruction = (
        "You are an expert university-level research professor. You are writing a specific section of a massive academic report. "
        "Do not summarize or cut corners. Expand heavily on your reasoning. "
        "Strictly adhere to the exact chart schemas requested below when asked.\n\n"
        "STANDARD CHART FORMAT:\n"
        "<chart>\n{{\"title\": \"Title\", \"type\": \"bar\", \"labels\": [\"A\", \"B\"], \"values\": [10, 20]}}\n</chart>\n"
        "TIMELINE CHART FORMAT:\n"
        "<chart>\n{{\"title\": \"Timeline\", \"type\": \"timeline\", \"events\": [{{\"date\": \"2020\", \"label\": \"Event1\"}}, {{\"date\": \"2021\", \"label\": \"Event2\"}}]}}\n</chart>\n"
        "AREA CHART FORMAT:\n"
        "<chart>\n{{\"title\": \"Area Chart\", \"type\": \"area\", \"labels\": [\"Q1\", \"Q2\", \"Q3\"], \"values\": [30, 40, 35]}}\n</chart>\n"
        "RADAR CHART FORMAT:\n"
        "<chart>\n{{\"title\": \"Metrics\", \"type\": \"radar\", \"categories\": [\"A\", \"B\", \"C\"], \"values\": [80, 75, 90]}}\n</chart>\n"
        "SUNBURST CHART FORMAT:\n"
        "<chart>\n{{\"title\": \"Hierarchy\", \"type\": \"sunburst\", \"labels\": [\"Total\", \"A\", \"B\"], \"parents\": [\"\", \"Total\", \"Total\"], \"values\": [100, 60, 40]}}\n</chart>\n"
        "DO NOT use markdown json tags inside the <chart> tags."
    )

    sections = [
        "Write PART 1: Abstract & Historical Framework for the topic: '{question}'. \n"
        "GOAL: 1000 words. \n"
        "REQUIREMENTS: Insert exactly ONE Markdown Table detailing historical literature or timelines. "
        "DO NOT include any <chart> elements in this section. \n\n"
        "Context Sources:\n{context}",
        
        "Write PART 2: Comprehensive Analytical Deep-Dive for the topic: '{question}'. \n"
        "GOAL: 1200 words. \n"
        "REQUIREMENTS: Provide heavy text analysis and insert exactly TWO Charts using the <chart> tags: "
        "One 'timeline' chart showing key historical events and One 'area' chart showing progression or trends. Separate them with at least 500 words of text. \n\n"
        "Context Sources:\n{context}",
        
        "Write PART 3: Statistical Insights & Quantitative Evaluation for the topic: '{question}'. \n"
        "GOAL: 1200 words. \n"
        "REQUIREMENTS: Insert exactly ONE Markdown Table showing statistical comparisons. "
        "Also insert exactly THREE Charts using the <chart> tags: One 'bar' chart, One 'pie' chart, and One 'line' chart. "
        "Separate each visual heavily with textual analysis. \n\n"
        "Context Sources:\n{context}",
        
        "Write PART 4: Structural Outlook, Future Extensions, and Final Summary for the topic: '{question}'. \n"
        "GOAL: 800 words. \n"
        "REQUIREMENTS: Insert exactly TWO powerful Charts using the <chart> tags on the SAME LINE: "
        "One 'radar' chart showing comprehensive research metrics and One 'sunburst' chart showing data hierarchy and breakdown. "
        "These must be separate chart tags. End with 4-5 high-impact bullet points summarizing the entire research scope. \n\n"
        "Context Sources:\n{context}"
    ]

    full_report = ""
    
    for i, section_prompt in enumerate(sections, 1):
        prompt = ChatPromptTemplate.from_messages([
            ("system", system_instruction),
            ("human", section_prompt)
        ])
        
        synthesizer = prompt | reasoning
        try:
            response = synthesizer.invoke({"context": context_str, "question": state.question})
            full_report += response.content + "\n\n---\n\n"
        except Exception as e:
            full_report += f"\n\n[Error generating section {i}: {e}]\n\n"

    return {"final_answer": full_report.strip(), "references": references}

# ==========================================
# WORKFLOW DEFINITION
# ==========================================

workflow = StateGraph(AgentState)
workflow.add_node("planner", plan_research_node)
workflow.add_node("search_engine", execute_search_node)
workflow.add_node("continue_research", continue_research_node)
workflow.add_node("finalizer", finalize_answer_node)

workflow.set_entry_point("planner")
workflow.add_edge("planner", "search_engine")
workflow.add_conditional_edges(
    "search_engine",
    evaluate_research_node,
    {"continue": "continue_research", "finalize": "finalizer"}
)
workflow.add_edge("continue_research", "search_engine")
workflow.add_edge("finalizer", END)

app = workflow.compile()

# ==========================================
# STREAMLIT UI & DASHBOARD RENDERER
# ==========================================

def render_inline_chart(chart_json_str: str, col=None):
    """Parses chart JSON and renders all chart types."""
    try:
        clean_str = chart_json_str.strip()
        if clean_str.startswith("```json"):
            clean_str = clean_str[7:]
        elif clean_str.startswith("```"):
            clean_str = clean_str[3:]
            
        if clean_str.endswith("```"):
            clean_str = clean_str[:-3]
            
        clean_str = clean_str.strip()

        chart = json.loads(clean_str)
        chart_type = chart.get("type", "bar").lower()
        
        container = col if col else st
        
        # Timeline chart
        if chart_type == "timeline":
            events = chart.get("events", [])
            dates = [e.get("date", "") for e in events]
            labels = [e.get("label", "") for e in events]
            
            fig = go.Figure()
            for i, (date, label) in enumerate(zip(dates, labels)):
                fig.add_trace(go.Scatter(
                    x=[date], y=[i],
                    mode='markers+text',
                    text=[label],
                    textposition="top center",
                    marker=dict(size=12, color="#0d233a"),
                    hoverinfo="text"
                ))
            
            fig.update_layout(
                title=chart.get("title", "Timeline"),
                xaxis_title="Date",
                yaxis=dict(showticklabels=False),
                height=350,
                showlegend=False,
                margin=dict(t=40, b=20, l=10, r=10)
            )
            container.plotly_chart(fig, use_container_width=True)
            return

        # Radar chart
        if chart_type == "radar":
            categories = chart.get("categories", [])
            values = chart.get("values", [])
            
            fig = go.Figure(data=go.Scatterpolar(
                r=values,
                theta=categories,
                fill='toself',
                name=chart.get("title", "Radar"),
                line=dict(color="#0d233a"),
                fillcolor="rgba(13, 35, 58, 0.3)"
            ))
            
            fig.update_layout(
                title=chart.get("title", "Radar Chart"),
                polar=dict(radialaxis=dict(visible=True, range=[0, max(values) if values else 100])),
                height=350,
                margin=dict(t=40, b=20, l=10, r=10)
            )
            container.plotly_chart(fig, use_container_width=True)
            return

        # Sunburst chart
        if chart_type == "sunburst":
            labels = chart.get("labels", [])
            parents = chart.get("parents", [])
            values = chart.get("values", [])
            
            fig = go.Figure(go.Sunburst(
                labels=labels,
                parents=parents,
                values=values,
                marker=dict(colorscale='Blues'),
                name=chart.get("title", "Sunburst")
            ))
            
            fig.update_layout(
                title=chart.get("title", "Sunburst Chart"),
                height=350,
                margin=dict(t=40, b=20, l=10, r=10)
            )
            container.plotly_chart(fig, use_container_width=True)
            return

        # Standard charts (bar, line, pie, area)
        df = pd.DataFrame({
            "Category": chart.get("labels", []),
            "Value": [float(v) for v in chart.get("values", [])]
        })
        
        if chart_type == "pie":
            fig = px.pie(df, values="Value", names="Category", title=chart.get("title"), hole=0.4)
        elif chart_type == "bar":
            fig = px.bar(df, x="Category", y="Value", title=chart.get("title"), color="Category")
        elif chart_type == "line":
            fig = px.line(df, x="Category", y="Value", title=chart.get("title"), markers=True)
        elif chart_type == "area":
            fig = px.area(df, x="Category", y="Value", title=chart.get("title"))
        else:
            st.warning(f"Unrecognized chart type: {chart_type}")
            return
        
        fig.update_layout(margin=dict(t=40, b=20, l=10, r=10), height=350)
        container.plotly_chart(fig, use_container_width=True)
        
    except json.JSONDecodeError as e:
        st.error("⚠️ Chart format error.")
        with st.expander("View Raw"):
            st.code(chart_json_str)
    except Exception as e:
        st.error(f"⚠️ Error: {e}")

def main_streamlit():
    st.set_page_config(page_title="Academic Analytics Engine", page_icon="🎓", layout="wide")
    
    st.markdown("""
        <style>
        .main {background-color: #f8f9fa;}
        .stButton > button { background-color: #0d233a; color: white; padding: 12px 24px; border-radius: 4px; font-weight: 600;}
        .stButton > button:hover { background-color: #1a365d; }
        table {width: 100%; margin-bottom: 30px; margin-top: 15px;}
        th {background-color: #f1f3f5; color: #343a40;}
        td {padding: 10px;}
        </style>
    """, unsafe_allow_html=True)
    
    st.markdown("# 🎓 Academic Deep-Research & Analytics Infrastructure")
    st.markdown("Generates comprehensive 4000+ word reports via sequential LLM generation pipelines.")
    
    with st.sidebar:
        st.header("🔎 Research Scope")
        research_topic = st.text_input("Topic Identifier", placeholder="e.g., Global logistics bottle-necks")
        st.info("💡 **Note:** Sequential 4-part generation takes roughly 2-3 minutes. Please be patient.")
        
    if st.button("Execute Comprehensive Evaluation", use_container_width=True):
        if not research_topic:
            st.error("❌ Action item required: Input a target research topic identifier.")
        else:
            with st.spinner("Executing iterative research and compiling 4-part sequential manuscript..."):
                try:
                    final_state = app.invoke({"question": research_topic})
                    output_text = final_state.get("final_answer", "")
                    
                    st.markdown("---")
                    st.markdown("## 📖 COMPREHENSIVE LITERARY MANUSCRIPT")
                    st.markdown("---")
                    
                    parts = re.split(r'<chart>\s*(.*?)\s*</chart>', output_text, flags=re.DOTALL)
                    
                    charts_rendered = 0
                    in_part4 = False
                    pending_chart = None
                    
                    for i, part in enumerate(parts):
                        if i % 2 == 0:
                            # Text section
                            if part.strip():
                                # Check if entering PART 4
                                if "PART 4" in part or "Structural Outlook" in part:
                                    in_part4 = True
                                st.markdown(part.strip())
                        else:
                            # Chart section
                            if in_part4 and pending_chart is None:
                                # First chart of PART 4 - hold it
                                pending_chart = part
                            elif in_part4 and pending_chart is not None:
                                # Second chart of PART 4 - render side-by-side
                                col1, col2 = st.columns(2)
                                render_inline_chart(pending_chart, col=col1)
                                render_inline_chart(part, col=col2)
                                charts_rendered += 2
                                pending_chart = None
                                in_part4 = False
                            else:
                                # Regular chart
                                render_inline_chart(part)
                                charts_rendered += 1
                    
                    # Handle leftover chart
                    if pending_chart:
                        render_inline_chart(pending_chart)
                        charts_rendered += 1
                    
                    st.markdown("---")
                    st.markdown("## 🌐 DOCUMENTED SOURCE REGISTRY")
                    for key in sorted(final_state["references"].keys()):
                        ref = final_state["references"][key]
                        with st.expander(f"Reference Element [{key}] — {ref['title'][:80]}..."):
                            st.write(f"**Title:** {ref['title']}")
                            if ref.get('url'):
                                st.markdown(f"**URL Route:** [{ref['url']}]({ref['url']})")
                    
                    st.markdown("---")
                    c1, c2 = st.columns(2)
                    with c1:
                        st.metric("Synthesized Document Length", f"{len(output_text.split())} words")
                    with c2:
                        st.metric("Charts Rendered", f"{charts_rendered} / 6")
                        
                except Exception as e:
                    st.error(f"Pipeline Interruption: {str(e)}")

if __name__ == "__main__":
    main_streamlit()
